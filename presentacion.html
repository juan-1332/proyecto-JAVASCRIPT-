export class GameApp extends HTMLElement {
  connectedCallback() {
    this.render();
    this.configurarEventos();
  }

  renderizarVistaActiva() {
    switch (this.vistaActual) {
      case 'deck':        return `<deck-selector></deck-selector>`;
      case 'battle':      return `<battle-arena></battle-arena>`;
      case 'leaderboard': return `<leaderboard-view></leaderboard-view>`;
      case 'admin':       return this.administradorActual
        ? `<cards-manager></cards-manager>`
        : `<admin-login></admin-login>`;
    }
  }
}
customElements.define('card-game-app', GameApp);

src/utils/battleEngine.js 
export function calcularDanoAccion(danoBase, defensora) {
  const factor = factorAleatorio(0.85, 1.15);
  const bruto = Math.round(danoBase * factor);
  const reduccion = defensora.defending ? 0.5 : 0;
  return Math.max(1, Math.round(bruto * (1 - reduccion)));
}

06 · Capa de Datos & Resiliencia

Cliente REST puro con respaldo automático.

apiconfig.js consume /api/* con fetch() y, si el proyecto se abre sin backend (Live Server o Vercel estático), conmuta de forma transparente a db.json + localStorage.

1. Detección de Entorno Lee VITE_API_MODE, VITE_API_DEV_URL y VITE_API_PROD_URL con switch.
2. Servidor Express + Vite (Local) server.js expone /api/cards, /players, /battles y /admins persistiendo directo en src/data/db.json.
3. Fallback Inteligente Si fetch() recibe 404 HTML o error de red, atenderRespaldoEstatico() emula los 5 verbos HTTP en el navegador.

        07 · Detalle de Ingeniería
        

          Motor de sonido 100% procedural.
        

        

          Sin depender de archivos MP3 externos: soundEngine.js sintetiza cada golpe, escudo y fanfarria en tiempo real con osciladores de la Web Audio API. Pruébalo aquí:
        


        

          
          
          
          
          
        


        

          

            

            src/utils/soundEngine.js
          


reproducirTono({ freq, endFreq, duration, type, gain }) {
  const osc = ctx.createOscillator();
  const amp = ctx.createGain();
  osc.type = type; // 'sawtooth' | 'triangle' | 'square' | 'sine'
  osc.frequency.setValueAtTime(freq, now);
  osc.frequency.exponentialRampToValueAtTime(endFreq, now + duration);
  osc.connect(amp); amp.connect(ctx.destination);
  osc.start(now); osc.stop(now + duration);
}
