Core Engine

ejecutarAccionTurno()

src/utils/battleEngine.js

Función pura que recibe estado + acción, calcula daño (factor aleatorio + escudo), gestiona KO, relevo de carta y avance de turno. Devuelve { nextState, event }.

export function ejecutarAccionTurno(estado, claveActor, accion) {
  if (estado.status !== 'playing' || estado.currentActor !== claveActor)
    return { nextState: estado, event: null };

  const siguienteEstado = structuredClone(estado);
  // ataque | defensa | especial → daño, logs, KO, swap
  siguienteEstado.globalTurn += 1;
  siguienteEstado.currentActor = claveOponente;
  return { nextState: siguienteEstado, event: evento };
}

solicitarApi intenta REST; si falla (Live Server / Vercel estático) cae a db.json + localStorage.

export async function solicitarApi(ruta, opciones = {}) {
  try {
    const respuesta = await fetch(url, configuracion);
    if (respuesta.ok) return await respuesta.json();
    return await atenderRespaldoEstatico(ruta, opciones);
  } catch (e) {
    if (/Failed to fetch/i.test(e.message))
      return await atenderRespaldoEstatico(ruta, opciones);
    throw e;
  }
}

Respaldo offline completo

¿Cómo ofrecer CRUD real (GET/POST/PUT/PATCH/DELETE + filtros + sort) cuando no hay backend (Live Server / Vercel estático)?

Solución

atenderRespaldoEstatico()

Simula un mini JSON-Server en el cliente: lee/escribe localStorage, aplica filter + sort y genera IDs. Transparente para el resto de la app.
